/** * SISTEM PEMARKAHAN PENTAKSIRAN BERTERUSAN GKKO1062 * IPG KAMPUS DARULAMAN - BACKEND SCRIPT V2 (STABIL & BEBAS RALAT CORS) */ function doGet(e) { try { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheetByName("data"); if (!sheet) { return ContentService.createTextOutput(JSON.stringify({ status: "error", error: "Helaian bernama 'data' tidak dijumpai!" })).setMimeType(ContentService.MimeType.JSON); } var range = sheet.getDataRange(); var values = range.getValues(); var students = []; for (var i = 1; i < values.length; i++) { var row = values[i]; if (!row[1] || row[1].toString().trim() === "") continue; students.push({ rowNum: i + 1, no: row[0] ? row[0].toString() : "", name: row[1] ? row[1].toString().toUpperCase() : "", kp: row[2] ? row[2].toString().trim() : "", group: row[3] ? row[3].toString().trim().toUpperCase() : "TIADA UNIT", t1: parseFloat(row[4]) || 0, t2: { k1: parseFloat(row[5]) || 0, k2: parseFloat(row[6]) || 0, k3: parseFloat(row[7]) || 0, k4: parseFloat(row[8]) || 0 }, t3: { k1: parseFloat(row[9]) || 0, k2: parseFloat(row[10]) || 0, k3: parseFloat(row[11]) || 0, k4: parseFloat(row[12]) || 0, k5: parseFloat(row[13]) || 0, k6: parseFloat(row[14]) || 0, k7: parseFloat(row[15]) || 0, k8: parseFloat(row[16]) || 0 }, total: parseFloat(row[17]) || 0, grade: row[18] ? row[18].toString().trim() : "" }); } return ContentService.createTextOutput(JSON.stringify({ status: "success", data: students })).setMimeType(ContentService.MimeType.JSON); } catch (error) { return ContentService.createTextOutput(JSON.stringify({ status: "error", error: error.toString() })).setMimeType(ContentService.MimeType.JSON); } } function doPost(e) { try { if (!e.postData || !e.postData.contents) { return ContentService.createTextOutput(JSON.stringify({ status: "error", error: "Tiada data diterima!" })).setMimeType(ContentService.MimeType.JSON); } var params = JSON.parse(e.postData.contents); var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheetByName("data"); if (!sheet) { return ContentService.createTextOutput(JSON.stringify({ status: "error", error: "Helaian bernama 'data' tidak dijumpai!" })).setMimeType(ContentService.MimeType.JSON); } var rowNum = parseInt(params.rowNum); if (!rowNum || rowNum < 2) { return ContentService.createTextOutput(JSON.stringify({ status: "error", error: "Nombor baris tidak sah!" })).setMimeType(ContentService.MimeType.JSON); } // Kemas kini Tugasan 1 sheet.getRange(rowNum, 5).setValue(params.t1); // Kemas kini Tugasan 2 (Kolom F hingga I) sheet.getRange(rowNum, 6).setValue(params.t2.k1); sheet.getRange(rowNum, 7).setValue(params.t2.k2); sheet.getRange(rowNum, 8).setValue(params.t2.k3); sheet.getRange(rowNum, 9).setValue(params.t2.k4); // Kemas kini Tugasan 3 (Kolom J hingga Q) sheet.getRange(rowNum, 10).setValue(params.t3.k1); sheet.getRange(rowNum, 11).setValue(params.t3.k2); sheet.getRange(rowNum, 12).setValue(params.t3.k3); sheet.getRange(rowNum, 13).setValue(params.t3.k4); sheet.getRange(rowNum, 14).setValue(params.t3.k5); sheet.getRange(rowNum, 15).setValue(params.t3.k6); sheet.getRange(rowNum, 16).setValue(params.t3.k7); sheet.getRange(rowNum, 17).setValue(params.t3.k8); // Pengiraan Skor Akhir var t2_total = parseFloat(params.t2.k1) + parseFloat(params.t2.k2) + parseFloat(params.t2.k3) + parseFloat(params.t2.k4); var t3_total = parseFloat(params.t3.k1) + parseFloat(params.t3.k2) + parseFloat(params.t3.k3) + parseFloat(params.t3.k4) + parseFloat(params.t3.k5) + parseFloat(params.t3.k6) + parseFloat(params.t3.k7) + parseFloat(params.t3.k8); var finalScore = parseFloat(params.t1) + t2_total + t3_total; finalScore = Math.round(finalScore * 100) / 100; sheet.getRange(rowNum, 18).setValue(finalScore); sheet.getRange(rowNum, 19).setValue(calculateGrade(finalScore)); SpreadsheetApp.flush(); return ContentService.createTextOutput(JSON.stringify({ status: "success", message: "Markah berjaya disimpan!" })).setMimeType(ContentService.MimeType.JSON); } catch (error) { return ContentService.createTextOutput(JSON.stringify({ status: "error", error: error.toString() })).setMimeType(ContentService.MimeType.JSON); } } function calculateGrade(score) { if (score >= 90) return 'A+'; if (score >= 80) return 'A'; if (score >= 75) return 'A-'; if (score >= 70) return 'B+'; if (score >= 65) return 'B'; if (score >= 60) return 'B-'; if (score >= 55) return 'C+'; if (score >= 50) return 'C'; if (score >= 40) return 'D'; return 'F'; }
